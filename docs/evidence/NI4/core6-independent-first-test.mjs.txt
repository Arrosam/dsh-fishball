// Independent canonical JSON oracle: never import the product serializer/digest.
function canonical(value) {
  if (Array.isArray(value)) return `[${value.map(canonical).join(',')}]`;
  if (value !== null && typeof value === 'object') {
    return `{${Object.keys(value).sort().map(key => `${JSON.stringify(key)}:${canonical(value[key])}`).join(',')}}`;
  }
  return JSON.stringify(value);
}
const canonicalDigest = value => digest(canonical(value));
function denied(code, operation) {
  assert.throws(operation, error => error?.code === code && error.message === code, code);
}

// Only schema/count/digest observations leave this helper. All SELECTs are read-only
// on our synthetic stores; no row plaintext or raw ciphertext is persisted/output.
function readOnlyFacts(path) {
  const db = new DatabaseSync(path, {readOnly: true});
  try {
    const tables = ['tasks', 'actions', 'reservations', 'runtime_commands', 'runtime_data',
      'runtime_records', 'automation_records', 'workspace_memory'];
    const counts = Object.fromEntries(tables.map(name =>
      [name, db.prepare(`SELECT count(*) AS n FROM ${name}`).get().n]));
    const hashRows = rows => rows.map(row => {
      const {nonce, ciphertext, tag, metadata, ...identity} = row;
      const hash = createHash('sha256');
      hash.update(canonical(identity));
      for (const part of [nonce, ciphertext, tag]) hash.update(part === null ? Buffer.alloc(0) : Buffer.from(part));
      hash.update(metadata);
      return {...identity, digest: hash.digest('hex')};
    });
    const data = hashRows(db.prepare('SELECT owner,workspace,task,id,nonce,ciphertext,tag,metadata FROM runtime_data ORDER BY owner,workspace,task,id').all());
    const memory = hashRows(db.prepare('SELECT owner,workspace,namespace,revision,nonce,ciphertext,tag,metadata FROM workspace_memory ORDER BY owner,workspace,namespace').all());
    return {schemaVersion: db.prepare('PRAGMA user_version').get().user_version, counts, data, memory};
  } finally { db.close(); }
}

      environmentRef: fixed('environment'), bindingDigest: fixed('environment-binding'),

    ...(executionInput ? {executionInput} : {})};
}
function approve(x, spec, expiresAt = 20000) {
  const plan = x.store.planAction(x.owner, {...x.bind(), spec});
  x.store.grantAction(x.owner, {...x.bind(), actionRef: plan.actionRef,
    actionDigest: plan.actionDigest, expiresAt});
  return {...x.bind(), actionRef: plan.actionRef, actionDigest: plan.actionDigest};
}

  const goal = x.store.listData(x.owner, {taskRef: x.task.taskRef, kind: 'goal'})[0];

  return x.store.configureAudit(x.owner, {...x.bind(), auditRef: ref('audit'), expectedAuditRevision: 0,
    artifactRef: selected.artifactRef, artifactVersion: selected.version, digest: selected.digest,
    contextRecordRef: context.recordRef, material: unique, policyRef: fixed('explicit-review-policy'),
    blockingSeverities: ['blocking'], maxRepairRounds: 2});
}
function reviewed(x, selected, {context = storedContext(x), material = [], response = {}} = {}) {

  const input = x.store.prepareAuditInput(x.owner, {...x.bind(), auditRef: requirement.auditRef,
    expectedAuditRevision: requirement.auditRevision, purpose: 'audit_review', round: 0});
  const generationRef = ref('review-generation'), sessionRef = ref('review-session');
  const spec = actionSpec(x, {kind: 'review', targetDigest: input.bindingDigest,
    dataDigest: selected.digest, usageUnits: 100000, runtimeMs: 100, mode: 'create',
    generationRef, sessionRef, executionInput: {recordRef: input.recordRef, digest: input.digest,
      purpose: input.purpose, auditRef: input.auditRef, bindingDigest: input.bindingDigest,
      materialDigest: input.materialDigest}});
  const held = x.store.claimAction(x.host, approve(x, spec));
  x.store.workerObservation(x.host, {taskRef: x.task.taskRef, commandRef: ref('created-command'),
    claimRef: held.claimRef, generationRef, sessionRef, phase: 'created', cursor: 0,
    evidenceRef: ref('created-evidence')});
  const reviewer = {};
  x.store.bindAuditReviewer(x.host, reviewer, {taskRef: x.task.taskRef,
    auditRef: requirement.auditRef, claimRef: held.claimRef, generationRef, sessionRef});
  const execution = x.store.beginAuditExecution(x.host, {...x.bind(), actionRef: held.actionRef,
    actionDigest: held.actionDigest, claimRef: held.claimRef, generationRef, inputRecordRef: input.recordRef});
  const body = {contractVersion: 'df.audit-response/v1', bindingDigest: input.bindingDigest,
    materialDigest: input.materialDigest, inputDigest: input.digest, verdict: 'passed',
    coverageRefs: requirement.material.map(item => item.recordRef), unexaminedRefs: [], findings: [],
    ...(typeof response === 'function' ? response(requirement) : response)};
  const capture = x.store.captureAuditResult(x.host, {taskRef: x.task.taskRef,
    commandRef: ref('capture-command'), claimRef: held.claimRef, generationRef,
    executionRef: execution.executionRef, sessionRef, inputCursor: 1, cursor: 2,

  x.store.submitAuditResult(reviewer, {taskRef: x.task.taskRef, commandRef: ref('submit-command'),

  x.store.workerObservation(x.host, {taskRef: x.task.taskRef, commandRef: ref('stop-command'),

    commandRef: ref('settlement-command'), claimRef: held.claimRef, generationRef,
    outcome: 'completed', evidenceRef: capture.evidenceRef});
  assert.equal(settled.reservationHeld, false);
  return {artifact: x.store.artifact(x.owner, {taskRef: x.task.taskRef, artifactRef: selected.artifactRef}),
    context, requirement, input, capture};
}

function publicationUnits(x, delivery, selected) {
  const content = x.store.readData(x.owner, {taskRef: x.task.taskRef, recordRef: selected.recordRef}).content;

    artifactRef: selected.artifactRef, artifactVersion: selected.version, digest: selected.digest,
    recipientRef: delivery.recipientRef, channelRef: delivery.channelRef, content};
  return Buffer.byteLength(canonical(envelope), 'utf8');
}

  test('C6F07 precommit/permission cuts preserve hold; expiry/revoke ordering keeps historical sent/current access separate', t => {

    const sent = x.store.publishLocalDelivery(x.host, request); assert.equal(sent.state, 'sent');

    for (const observation of [{observedAt: 4099, permissionState: 'allowed', code: 'STALE_OBSERVATION'},
