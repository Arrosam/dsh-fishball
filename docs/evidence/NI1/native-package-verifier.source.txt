from pathlib import Path, PurePosixPath
import hashlib, json, tarfile, sys, datetime
base=Path(__file__).resolve().parent
source=base/'source'
artifact=base/'artifacts/dsh-fishball-0.1.0-preview.1.tgz'
package=base/'home/profiles/web/node_modules/dsh-fishball'
freeze=Path(sys.argv[1]).resolve()
source_commit=sys.argv[2]
sha=lambda b:hashlib.sha256(b).hexdigest()
assert len(source_commit)==40 and set(source_commit)<=set('0123456789abcdef')
assert source.is_dir() and artifact.is_file() and not package.exists()
files={}
with tarfile.open(artifact,'r:gz') as archive:
 for member in archive.getmembers():
  path=PurePosixPath(member.name)
  assert not path.is_absolute() and '..' not in path.parts and path.parts[0]=='package'
  if member.isdir(): continue
  assert member.isfile(), 'NONREGULAR_PACKAGE_MEMBER'
  relative=str(PurePosixPath(*path.parts[1:]))
  assert relative not in files and relative not in ['', '.']
  assert not any(part in {'.git','.runtime','node_modules','.aws','.codex','.agents'} for part in path.parts)
  assert not any(part.endswith(('.sqlite','.db','.tgz','.backup')) or part in {'key.bin','state.json','native-access.json'} for part in path.parts)
  content=archive.extractfile(member).read()
  assert (source/relative).is_file() and content==(source/relative).read_bytes(), 'ARCHIVE_SOURCE_MISMATCH'
  files[relative]=content
public=[p.relative_to(source).as_posix() for folder in ['src','scripts','lib'] for p in (source/folder).rglob('*') if p.is_file()] + ['package.json','LICENSE','THIRD_PARTY_NOTICES.md']
assert len(public)==len(set(public))
for relative in public: assert relative in files and files[relative]==(source/relative).read_bytes()
held=[]
for line in freeze.read_text().splitlines():
 digest,relative=line.split('  ',1)
 assert relative in files and sha(files[relative])==digest, 'EXECUTABLE_FREEZE_MISMATCH'
 held.append(relative)
package.mkdir(parents=True)
for relative,content in files.items():
 target=package/relative
 target.parent.mkdir(parents=True,exist_ok=True)
 target.write_bytes(content)
 target.chmod(0o644)
assert not any(p.is_symlink() for p in package.rglob('*'))
metadata=json.loads(files['package.json'])
assert metadata['version']=='0.1.0-preview.1' and metadata['private'] is True
assert metadata['exports']['./installed']=='./src/installed-plugin.mjs'
assert len(metadata['peerDependencies'])==21
report={'result':'passed','sourceCommit':source_commit,'artifact':str(artifact.relative_to(base.parent.parent)),'artifactSha256':sha(artifact.read_bytes()),'artifactBytes':artifact.stat().st_size,'archiveFiles':len(files),'publicFilesCompared':len(public),'publicFilesMatch':True,'regressionFreezeFilesMatched':len(held),'physicalNormalPackage':True,'pluginRelativeRuntimeAbsent':not (package/'.runtime').exists(),'sourceCheckoutLink':False,'ownerProfileUsed':False,'ownerCredentialsConfigured':False,'offline':True,'ignoreScripts':True,'packageVersion':metadata['version'],'consumedPinnedPeers':len(metadata['peerDependencies']),'clientBundleSha256':sha(files['lib/client.js']),'profileEntry':'dsh-fishball/installed','timestamp':datetime.datetime.now(datetime.timezone.utc).isoformat()}
(base/'evidence/package-artifact.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report))
