// Stop only the actual isolated native child observed for this fixture.
import assert from 'node:assert/strict';
import { existsSync,readFileSync,writeFileSync } from 'node:fs';
import { execFileSync } from 'node:child_process';
import { fileURLToPath } from 'node:url';
import { setTimeout } from 'node:timers/promises';
const base=fileURLToPath(new URL('./',import.meta.url));
const pid=Number(process.argv[2]),cycle=process.argv[3];
try{
  assert.ok(Number.isSafeInteger(pid)&&pid>1);assert.ok(['initial','restart','uninstall','removed-profile','first-failure'].includes(cycle));
  const before=JSON.parse(readFileSync(`${base}evidence/observer-state.json`,'utf8'));assert.equal(before.pid,pid);
  const command=execFileSync('/bin/ps',['-p',String(pid),'-o','command='],{encoding:'utf8'});
  assert.ok(command.includes('/.runtime/portable-install/host/node_modules/@deepseek-ai/dsh/lib/bin.js'));
  assert.ok(command.includes('--profile web --host 127.0.0.1 --port 3080 --no-open'));
  process.kill(pid,'SIGTERM');
  let gone=false;const deadline=Date.now()+15000;
  while(Date.now()<deadline){try{process.kill(pid,0);}catch(error){if(error.code==='ESRCH'){gone=true;break;}throw error;}await setTimeout(50);}
  assert.equal(gone,true,'OWNED_NATIVE_CHILD_NOT_EXITED');
  const after=JSON.parse(readFileSync(`${base}evidence/observer-state.json`,'utf8'));assert.equal(after.pid,pid);assert.equal(after.phase,'disposed');
  assert.equal(existsSync(`${base}private/native-access.json`),false);
  for(const name of ['sessions','providers','inferenceAttempts','toolAttempts'])assert.equal(after[name],0);
  let listener=false;try{execFileSync('/usr/sbin/lsof',['-nP','-iTCP:3080','-sTCP:LISTEN'],{stdio:['ignore','pipe','pipe']});listener=true;}catch(error){assert.equal(error.status,1);}
  assert.equal(listener,false);
  const report={result:'passed',cycle,pid,ownedNativeChildExited:true,observerDisposed:true,temporaryAccessRemoved:true,listener3080Absent:true,databaseRetained:after.databaseExists,setupRetained:after.setupNamespaceExists,setupCleanupCode:after.setupCleanupCode??null,runtimeCleanupCode:after.runtimeCleanupCode??null,cleanupObservation:'ancestor cleanup ordering only; own unload fencing separately tested',sessions:0,providers:0,inferenceAttempts:0,toolAttempts:0,timestamp:new Date().toISOString()};
  writeFileSync(`${base}evidence/${cycle}-stop.json`,JSON.stringify(report,null,2)+'\n',{mode:0o600});console.log(JSON.stringify(report));
}catch{console.error('NORMAL_NATIVE_OWNED_STOP_FAILED');process.exitCode=1;}
