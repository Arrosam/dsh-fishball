import { writeFileSync, renameSync, unlinkSync, existsSync } from 'node:fs';
import { fileURLToPath } from 'node:url';
const base=fileURLToPath(new URL('./',import.meta.url));
const statePath=`${base}evidence/observer-state.json`, accessPath=`${base}private/native-access.json`;
export const name='df-normal-install-observer';
export const inject=['sessions','approval','agents','llm','tools','connection','webServer','loader'];
export function apply(ctx, config) {
  const state={phase:'observer-active',pid:process.pid,pluginEntryName:config.pluginEntryName,pluginAvailable:false,loads:0,cycles:[],sessions:ctx.sessions.list().length,providers:ctx.llm.listProviders().length,inferenceAttempts:0,toolAttempts:0,host:ctx.webServer.host,port:ctx.webServer.port};
  let borrowed, currentService, pending, active=true;
  const write=()=>{writeFileSync(`${statePath}.tmp`,JSON.stringify(state,null,2)+'\n',{mode:0o600});renameSync(`${statePath}.tmp`,statePath);};
  const identityProbe=service=>{
    try {service.intake({taskId:'identity-denial-probe',provider:'normal-install',source:'local-validation',eventId:'closed-input'});return 'unexpected-admission';}
    catch(error){return error.code??'UNAVAILABLE';}
  };
  const staleProbe=service=>{
    try {service.diagnostics();return 'unexpected-active';}
    catch(error){return error.code??'UNAVAILABLE';}
  };
  ctx.on('agent/request',()=>{state.inferenceAttempts++;write();throw new Error('NORMAL_INSTALL_INFERENCE_DISABLED');});
  ctx.tools.guard(()=>{state.toolAttempts++;write();return 'NORMAL_INSTALL_TOOLS_DISABLED';});
  writeFileSync(accessPath,JSON.stringify({url:ctx.connection.authenticatedUrl(`http://127.0.0.1:${ctx.webServer.port}`)})+'\n',{mode:0o600});
  write();
  ctx.inject(['df'], child=>{
    borrowed=child.df;currentService=borrowed;state.pluginAvailable=true;state.loads++;
    child.effect(()=>()=>{currentService=undefined;state.pluginAvailable=false;write();},'df.normal-install-presence');
    try {state.diagnostic=borrowed.diagnostics();state.pluginState='active';}
    catch(error){state.pluginState='diagnostic-unavailable';state.diagnosticCode=error.code??'UNAVAILABLE';}
    state.identityProbe=identityProbe(borrowed);write();
    if(config.lifecycleCycles && !pending) pending=new Promise(resolve=>setImmediate(resolve)).then(async()=>{
      const entries=[...ctx.loader.entries()].filter(entry=>entry.options.id==='df-installed');
      if(entries.length!==1 || entries[0].options.name!==config.pluginEntryName)throw new Error('NORMAL_INSTALL_ENTRY_SELECTION');
      const entry=entries[0];
      for(let i=0;i<config.lifecycleCycles && active;i++){
        const previous=borrowed;
        await entry.update({disabled:true});
        const staleReference=staleProbe(previous), serviceRemoved=currentService===undefined;
        await entry.update({disabled:false});await ctx.loader.await();
        const current=currentService, diagnostic=current.diagnostics(), identity=identityProbe(current);
        const cycle={cycle:i+1,staleReference,serviceRemoved,supported:diagnostic.supported,identityProbe:identity,sessions:ctx.sessions.list().length,providers:ctx.llm.listProviders().length};
        state.cycles.push(cycle);write();
        if(staleReference!=='SERVICE_DISPOSED'||!serviceRemoved||!diagnostic.supported||identity!=='IDENTITY_UNAVAILABLE'||cycle.sessions!==0||cycle.providers!==0)throw new Error('NORMAL_INSTALL_LIFECYCLE_ORACLE');
      }
      state.phase='cycles-completed';state.databaseRetained=existsSync(config.databasePath);write();
    }).catch(()=>{state.phase='lifecycle-failed';state.lifecycleCode='NORMAL_INSTALL_LIFECYCLE_FAILED';write();});
  });
  ctx.effect(()=>()=>{
    active=false;state.phase='disposed';state.staleReference=borrowed?staleProbe(borrowed):'unavailable';
    try {state.sessions=ctx.sessions.list().length;state.sessionObservation='queried-at-disposal';}
    catch {state.sessionObservation='host-service-already-disposed';}
    try {unlinkSync(accessPath);}catch{}
    write();
  },'df.normal-install-observer');
}
