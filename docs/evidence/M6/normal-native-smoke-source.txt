async function waitFor(check,code){const deadline=Date.now()+15000;let result;while(Date.now()<deadline){try{result=check();if(result)return result;}catch{}await setTimeout(50);}throw new Error(code);}
function persisted(){
  const names=readdirSync(vault).sort();assert.deepEqual(names,['key.bin','state.json']);
  const stats=[vault,database,`${vault}/key.bin`,`${vault}/state.json`].map(path=>lstatSync(path));
  assert.equal(stats[0].mode&0o777,0o700);
  for(const s of stats.slice(1)){assert.equal(s.mode&0o777,0o600);assert.equal(s.nlink,1);assert.equal(s.isSymbolicLink(),false);}
  const key=readFileSync(`${vault}/key.bin`);assert.equal(key.length,32);
  const snapshot={key:digest(key),metadata:digest(readFileSync(`${vault}/state.json`)),databaseInode:String(stats[1].ino)};
  key.fill(0);return snapshot;
}
async function request(url,options={}){try{return await fetch(url,{...options,redirect:'manual',signal:AbortSignal.timeout(5000)});}catch{throw new Error('NORMAL_NATIVE_HTTP_UNAVAILABLE');}}
async function ownerCalls(){
  const access=JSON.parse(readFileSync(`${base}private/native-access.json`,'utf8')),url=new URL(access.url);
  assert.equal(url.origin,origin);assert.equal(url.pathname,'/');assert.equal(url.searchParams.size,1);assert.equal(url.searchParams.has('token'),true);
  const exchange=await request(url.href);assert.equal(exchange.status,303);assert.equal(exchange.headers.get('location'),'./');
  let cookie=exchange.headers.get('set-cookie')?.split(';')[0];assert.ok(cookie);
  return {
    async routeStatus(channel,mode='owner'){
      const response=await request(`${origin}/${channel}/v1/status`,{method:'POST',headers:{'content-type':'application/json',origin,...(mode==='anonymous'?{}:{cookie})},body:JSON.stringify({type:'client-request',rpcId:'normal-uninstalled-readback',method:'v1/status',payload:{}})});return response.status;
    },
    async call(channel,endpoint,payload={},mode='owner',expectOk=true){
      const response=await request(`${origin}/${channel}/v1/${endpoint}`,{method:'POST',headers:{'content-type':'application/json',origin:mode==='cross-origin'?'http://synthetic.invalid':origin,...(mode==='anonymous'?{}:{cookie:mode==='forged'?'df-synthetic=forged':cookie})},body:JSON.stringify({type:'client-request',rpcId:'normal-install-synthetic',method:`v1/${endpoint}`,payload})});
      if(mode!=='owner')return response.status;
      assert.equal(response.status,200,'NATIVE_OWNER_RPC_STATUS');
      const result=(await response.json()).result;assert.ok(result&&typeof result==='object','NATIVE_RPC_ENVELOPE');
      const encoded=JSON.stringify(result);assert.equal(encoded.includes(cookie),false);assert.equal(encoded.includes(url.searchParams.get('token')),false);
      if(expectOk){assert.equal(result.ok,true,`NATIVE_OPERATION_${channel}_${endpoint}_${result.error?.code??'INVALID'}`);return result.value;}
      assert.equal(result.ok,false);assert.equal(Object.hasOwn(result,'value'),false);return result.error?.code;
    },
    clear(){cookie=undefined;}
  };
}
async function taskRoundtrip(call,cycle){
  const existing=await call('df-rpc','list');assert.equal(existing.length,cycle==='initial'?0:1);
  const input={taskId:'normal-installed-synthetic-task',provider:'synthetic-native-http',source:'normal-installed-synthetic',eventId:'bounded-fixed-intake',goal:'Synthetic normal install encrypted task; no inference, tools or external effects.'};
  const first=await call('df-rpc','intake',{...input,commandRef:digest(`normal-installed-${cycle}-first`)});assert.equal(first.deduplicated,cycle!=='initial');
  const duplicate=await call('df-rpc','intake',{...input,commandRef:digest(`normal-installed-${cycle}-duplicate`)});assert.equal(duplicate.deduplicated,true);
  let snapshot=await call('df-rpc','status',{taskRef:first.task.taskRef});
  const goal=snapshot.data.find(record=>record.kind==='goal');assert.ok(goal);
  const detail=await call('df-rpc','detail',{taskRef:first.task.taskRef,recordRef:goal.recordRef,expectedTaskVersion:snapshot.task.version,expectedAuthEpoch:snapshot.task.authEpoch});assert.equal(detail.data.content,input.goal);
  if(cycle==='initial')await call('df-rpc','control',{taskRef:first.task.taskRef,commandRef:digest('normal-installed-pause'),expectedTaskVersion:snapshot.task.version,expectedAuthEpoch:snapshot.task.authEpoch,kind:'pause'});
  snapshot=await call('df-rpc','status',{taskRef:first.task.taskRef});assert.equal(snapshot.control,'paused');assert.equal(snapshot.task.status,'queued');
  for(const name of ['actions','workers','executions'])assert.equal(snapshot[name].length,0);
  assert.equal((await call('df-rpc','list')).length,1);
  const key=readFileSync(`${vault}/key.bin`),publicRead=JSON.stringify({first,duplicate,snapshot,detail});
  assert.equal(publicRead.includes(key.toString('hex')),false);assert.equal(publicRead.includes(key.toString('base64')),false);
  for(const path of [database,`${database}-wal`,`${database}-shm`,`${vault}/state.json`])if(existsSync(path)){
    const raw=readFileSync(path);assert.equal(raw.includes(Buffer.from(input.goal)),false);assert.equal(raw.includes(key),false);
  }
  key.fill(0);closed();return {taskRef:first.task.taskRef,firstDeduplicated:first.deduplicated,duplicateDeduplicated:true,entries:1,control:snapshot.control,status:snapshot.task.status,actions:0,workers:0,executions:0,encryptedGoalAbsent:true,privateKeyExcluded:true};
}
async function lifecycle(operation,id){
  writeFileSync(`${base}private/lifecycle-command.json.tmp`,JSON.stringify({operation,id})+'\n',{mode:0o600});renameSync(`${base}private/lifecycle-command.json.tmp`,`${base}private/lifecycle-command.json`);
  const result=await waitFor(()=>{const current=state().lifecycle;return current?.id===id&&current.result?current:null;},'NORMAL_NATIVE_LIFECYCLE_TIMEOUT');
  assert.equal(result.result,'passed',result.code??'NORMAL_NATIVE_LIFECYCLE_FAILED');closed();return result;
}

  stage='native-owner-auth';
