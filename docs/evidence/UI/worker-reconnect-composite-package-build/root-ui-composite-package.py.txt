"""Build one private QA archive from frozen a58 bytes and two reviewed client files."""
from pathlib import Path, PurePosixPath
import copy
import gzip
import hashlib
import io
import json
import os
import tarfile

ROOT = Path('/Users/samuel/Documents/Codex/2026-10-04/task-6/dsh-fishball')
BASE = ROOT / '.runtime/ni2-install/artifact/dsh-fishball-0.1.0-preview.1.tgz'
BASE_SHA = 'fa3fb398d901a5c3d04d1f6f949878eb190f1d1fb737b16210347963d0afb5c0'
DEST = ROOT / '.runtime/ui-qa-client-cb75'
INSTALLED = ROOT / '.runtime/ui-qa-a58c522/home/profiles/web/node_modules/dsh-fishball'
REPLACEMENTS = {
    'src/client-entry.cjs': '4ba377f9973eddd6d1aebe66e014370f9f8ff7b1392b95528a51790040fd9e82',
    'lib/client.js': '9ba97b8397153dc29f87f1bcd6018cce856decdd485517399ea255cbf90792d8',
}
sha = lambda value: hashlib.sha256(value).hexdigest()

assert ROOT.resolve() == ROOT and BASE.resolve() == BASE
assert INSTALLED.resolve() == INSTALLED and INSTALLED.is_dir()
assert DEST.resolve() == DEST
assert not BASE.is_symlink() and sha(BASE.read_bytes()) == BASE_SHA
assert not DEST.exists()
replacement_bytes = {}
for name, expected in REPLACEMENTS.items():
    p = ROOT / name
    assert not p.is_symlink() and p.resolve().is_relative_to(ROOT)
    replacement_bytes[name] = p.read_bytes()
    assert sha(replacement_bytes[name]) == expected

records = []
members = []
seen = set()
unpacked_bytes = 0
with tarfile.open(BASE, mode='r:gz') as old:
    for member in old.getmembers():
        parts = PurePosixPath(member.name).parts
        assert parts and parts[0] == 'package'
        assert not member.name.startswith('/') and '..' not in parts and '.' not in parts
        assert member.name == '/'.join(parts), 'Noncanonical archived path'
        assert member.name not in seen
        seen.add(member.name)
        assert member.isfile() or member.isdir(), 'Links and special entries are forbidden'
        assert 0 <= member.size <= 32 * 1024 * 1024
        if member.isdir():
            members.append((copy.copy(member), None))
            continue
        name = '/'.join(parts[1:])
        assert name
        stream = old.extractfile(member)
        assert stream is not None
        before = stream.read()
        assert len(before) == member.size
        unpacked_bytes += len(before)
        assert unpacked_bytes <= 128 * 1024 * 1024
        installed = INSTALLED / name
        assert not installed.is_symlink() and installed.resolve().is_relative_to(INSTALLED.resolve())
        assert installed.read_bytes() == before, 'Original installed package drift'
        after = replacement_bytes.get(name, before)
        item = copy.copy(member)
        item.pax_headers = dict(member.pax_headers)
        item.size = len(after)
        if 'size' in item.pax_headers:
            item.pax_headers['size'] = str(len(after))
        members.append((item, after))
        records.append({'path': name, 'beforeBytes': len(before), 'beforeSha256': sha(before),
                        'bytes': len(after), 'sha256': sha(after), 'mode': member.mode})

assert len(records) == 1114
assert set(REPLACEMENTS).issubset({item['path'] for item in records})
assert [item['path'] for item in records if item['beforeSha256'] != item['sha256']] == [
    item['path'] for item in records if item['path'] in REPLACEMENTS]
assert sum(item['beforeSha256'] != item['sha256'] for item in records) == 2
actual_installed = set()
for p in INSTALLED.rglob('*'):
    assert not p.is_symlink(), 'Original installation may not contain links'
    if p.is_file():
        actual_installed.add(p.relative_to(INSTALLED).as_posix())
assert actual_installed == {item['path'] for item in records}

DEST.mkdir(mode=0o700)
package = DEST / 'package'
package.mkdir(mode=0o700)
archive = DEST / 'dsh-fishball-0.1.0-preview.1-ui-cb75.tgz'
with archive.open('xb') as raw, gzip.GzipFile(fileobj=raw, mode='wb', filename='', mtime=0) as zipped:
    with tarfile.open(fileobj=zipped, mode='w', format=tarfile.PAX_FORMAT) as new:
        for item, data in members:
            new.addfile(item, io.BytesIO(data) if data is not None else None)
            if data is not None:
                p = DEST / item.name
                assert p.resolve().is_relative_to(package)
                p.parent.mkdir(mode=0o700, parents=True, exist_ok=True)
                with p.open('xb') as output:
                    output.write(data)
                os.chmod(p, item.mode & 0o777)

with tarfile.open(archive, mode='r:gz') as built:
    files = {m.name.removeprefix('package/'): m for m in built.getmembers() if m.isfile()}
    assert set(files) == {item['path'] for item in records}
    for record in records:
        extracted = built.extractfile(files[record['path']])
        assert extracted is not None
        assert sha(extracted.read()) == record['sha256']
        assert sha((package / record['path']).read_bytes()) == record['sha256']

metadata = json.loads((package / 'package.json').read_text())
assert metadata['name'] == 'dsh-fishball' and metadata['version'] == '0.1.0-preview.1'
manifest = {'identity': 'a58 native package base / cb75 reviewed client only',
            'baseRevision': 'a58c522f32dd2be2ec29a697ec32ed295e154614',
            'clientRevision': 'cb75fdfd97842c89c78d8e96b8c4de1804261ed7',
            'baseArchiveSha256': BASE_SHA, 'archive': str(archive.relative_to(ROOT)),
            'archiveBytes': archive.stat().st_size, 'archiveSha256': sha(archive.read_bytes()),
            'files': records, 'fileCount': len(records), 'changedPaths': list(REPLACEMENTS),
            'wholeGitRevisionClaim': False, 'installedOldPackageStillUnchanged': True,
            'listenerProfileStorageAuthEffects': 0, 'nativeModelNetworkEffects': 0,
            'all87': 'HOLD', 'CODE': 'BLOCKED_UNCHANGED'}
with (DEST / 'composite-package-manifest.json').open('x') as output:
    output.write(json.dumps(manifest, indent=2, sort_keys=True) + '\n')
print(json.dumps({k: manifest[k] for k in ['identity', 'archive', 'archiveBytes',
                                         'archiveSha256', 'fileCount', 'changedPaths']}))
