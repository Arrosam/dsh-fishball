"""One fixed reviewed native sequence; retain private output, never retry."""
import datetime
import hashlib
import json
import os
from pathlib import Path
import subprocess
import time

ROOT = Path('/Users/samuel/Documents/Codex/2026-10-04/task-6/dsh-fishball')
CAPSULE = ROOT / '.runtime/ollama-dsh-functional-fixtures-r1'
NODE = Path('/opt/homebrew/opt/node@22/bin/node')
NODE_RESOLVED = Path('/opt/homebrew/Cellar/node@22/22.22.2_1/bin/node')
NODE_SHA = '463b47d6e984e4f84cd2421dabf5081414e40ab9d967e828b2db7858e707f6de'
sha = lambda value: hashlib.sha256(value).hexdigest()
now = lambda: datetime.datetime.now(datetime.timezone.utc).isoformat()
assert Path.cwd() == ROOT
assert NODE.resolve() == NODE_RESOLVED and sha(NODE.read_bytes()) == NODE_SHA
go_path = CAPSULE / 'root-execution-go.json'
go_bytes = go_path.read_bytes()
go = json.loads(go_bytes)
assert go['executionAuthorized'] is True and go['maxCalls'] == 6
assert go['endpoint'] == 'http://127.0.0.1:11434/api/chat'
assert go['model'] == 'deepseek-v4.1-flash:cloud'
assert go['authoritativeInferenceCountBefore'] == 0
assert go_path.stat().st_mode & 0o077 == 0
freeze_path = CAPSULE / 'fixture-freeze.json'
freeze_bytes = freeze_path.read_bytes()
assert sha(freeze_bytes) == go['fixtureFreezeSha256']
freeze = json.loads(freeze_bytes)
graph_path = ROOT / '.runtime/ni5-result-lifetime-positive/public-graph-pins.json'
graph = json.loads(graph_path.read_bytes())

def verify():
    failures = []
    for member in freeze['files']:
        raw = Path(member['path']).read_bytes()
        if len(raw) != member['bytes'] or sha(raw) != member['sha256']:
            failures.append(member['path'])
    for path, expected in go['supportingPins'].items():
        if sha((ROOT / path).read_bytes()) != expected:
            failures.append(path)
    base = Path(graph['root'])
    for path, expected in graph['files'].items():
        if sha((base / path).read_bytes()) != expected:
            failures.append(path)
    for path, expected in graph['links'].items():
        actual = base / path
        if not actual.is_symlink() or os.readlink(actual) != expected['target'] or str(actual.resolve()) != expected['resolved']:
            failures.append(path)
    return failures

assert not verify(), 'source drift before the sole run'
directory = ROOT / 'docs/evidence/NI3/ollama-six-slot-r1-live-execution'
directory.mkdir(mode=0o700)
private = CAPSULE / 'private-process-output'
private.mkdir(mode=0o700)
record = {'contract': 'df.root-ollama-six-slot-execution/v1', 'startedAt': now(),
    'argv': [str(NODE), str(CAPSULE / 'entry.mjs')], 'rootGoSha256': sha(go_bytes),
    'sourceFreezeSha256': sha(freeze_bytes), 'model': go['model'], 'maxCalls': 6,
    'attemptCountBefore': 0, 'outerProcessLimitSeconds': 495, 'retryAllowed': False,
    'sourceFilesChecked': len(freeze['files']), 'wholeHostFiles': len(graph['files']),
    'wholeHostLinks': len(graph['links']), 'all87': 'HOLD', 'CODE': 'BLOCKED_UNCHANGED'}
def save():
    (directory / 'root-run-record.json').write_text(json.dumps(record, indent=2, sort_keys=True) + '\n')
save()
started = time.monotonic()
def private_output(name):
    return os.fdopen(os.open(private / name, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600), 'wb')
with private_output('stdout.txt') as stdout, private_output('stderr.txt') as stderr:
    child = subprocess.Popen(record['argv'], cwd=ROOT, stdin=subprocess.DEVNULL,
        stdout=stdout, stderr=stderr, shell=False)
    record['childPid'] = child.pid
    save()
    try:
        record['exitCode'] = child.wait(timeout=495)
        record['outerTimeout'] = False
    except subprocess.TimeoutExpired:
        record['outerTimeout'] = True
        child.terminate()
        try:
            child.wait(timeout=5)
        except subprocess.TimeoutExpired:
            child.kill()
            child.wait(timeout=5)
        record['exitCode'] = child.returncode
        record['localProcessTermination'] = 'outer-bound-only'
        record['remoteTermination'] = 'unproved'
record['finishedAt'] = now()
record['durationSeconds'] = round(time.monotonic() - started, 6)
record['afterSourceDrift'] = verify()
record['inputsUnchanged'] = not record['afterSourceDrift']
record['privateOutput'] = {name: {'bytes': (private / name).stat().st_size,
    'sha256': sha((private / name).read_bytes())} for name in ['stdout.txt', 'stderr.txt']}
record['scopeNote'] = 'Read per-position observations; exit0 is not six passes or remote termination/settlement proof.'
save()
print(json.dumps({key: record[key] for key in ['exitCode', 'durationSeconds', 'outerTimeout', 'inputsUnchanged', 'privateOutput']}), flush=True)
