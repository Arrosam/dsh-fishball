"""Execute one root-reviewed finite plan, retaining immutable byte evidence."""
import datetime
import hashlib
import json
import os
from pathlib import Path
import shutil
import subprocess
import sys
import time

ROOT = Path('/Users/samuel/Documents/Codex/2026-10-04/task-6/dsh-fishball')
NODE = Path('/opt/homebrew/opt/node@22/bin/node')
NODE_RESOLVED = Path('/opt/homebrew/Cellar/node@22/22.22.2_1/bin/node')
NODE_HASH = '463b47d6e984e4f84cd2421dabf5081414e40ab9d967e828b2db7858e707f6de'
sha = lambda value: hashlib.sha256(value).hexdigest()
now = lambda: datetime.datetime.now(datetime.timezone.utc).isoformat()

def save(path, value):
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + '\n')

def source_path(name):
    path = Path(name) if name.startswith('/') else ROOT / name
    assert path.resolve() == NODE_RESOLVED or path.resolve().is_relative_to(ROOT), name
    return path

def verify(pins, graph):
    errors = []
    for name, expected in pins.items():
        try:
            if sha(source_path(name).read_bytes()) != expected:
                errors.append(name)
        except OSError as error:
            errors.append({'path': name, 'error': error.__class__.__name__})
    if graph is not None:
        base = Path(graph['root'])
        assert base.resolve().is_relative_to(ROOT)
        for name, expected in graph['files'].items():
            if sha((base / name).read_bytes()) != expected:
                errors.append(name)
        for name, value in graph['links'].items():
            path = base / name
            if not path.is_symlink() or os.readlink(path) != value['target'] or str(path.resolve()) != value['resolved']:
                errors.append(name)
    return errors

plan_path = source_path(sys.argv[1])
plan = json.loads(plan_path.read_text())
assert plan['rootReviewedExecutionGO'] is True and plan['runCountAuthorized'] == 1
assert plan['all87'] == 'HOLD' and plan['CODE'] == 'BLOCKED_UNCHANGED'
assert NODE.resolve() == NODE_RESOLVED
pins = dict(plan['pins'])
pins[str(NODE_RESOLVED)] = NODE_HASH
pins[str(plan_path.relative_to(ROOT))] = sha(plan_path.read_bytes())
runner = Path(__file__).resolve()
pins[str(runner.relative_to(ROOT))] = sha(runner.read_bytes())
graph = None
if plan.get('publicGraphManifest'):
    graph_path = source_path(plan['publicGraphManifest'])
    assert str(graph_path.relative_to(ROOT)) in pins
    graph = json.loads(graph_path.read_text())
directory = source_path(plan['evidenceDirectory'])
assert directory.is_relative_to(ROOT / 'docs/evidence') and not directory.exists()
directory.mkdir(mode=0o700, parents=True)
commands = plan['commands']
assert commands and len(commands) <= 16
for argv in commands:
    assert argv[0] == str(NODE) and len(argv) <= 12
    assert '--check' in argv or '--test' in argv
    assert all(isinstance(value, str) for value in argv)
freeze = dict(plan)
freeze.update({'contract': 'df.root-bounded-execution/v2', 'createdAt': now(),
    'root': str(ROOT), 'resolvedBinary': str(NODE_RESOLVED), 'pins': dict(sorted(pins.items())),
    'wholeGraphFiles': len(graph['files']) if graph else 0,
    'wholeGraphLinks': len(graph['links']) if graph else 0,
    'stopAfterFirstNonzeroCommand': True, 'timeoutSecondsPerCommand': 60})
save(directory / 'root-execution-freeze.json', freeze)
before = verify(pins, graph)
save(directory / 'root-before-verification.json', {'checkedAt': now(), 'inputCount': len(pins),
    'errors': before, 'fullOriginalGraphVerified': graph is not None})
assert not before, before
for name in plan.get('sourceCopies', []):
    assert name in pins and not name.startswith('/')
    destination = directory / 'frozen-source' / name.removeprefix('.runtime/')
    destination.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(source_path(name), destination)
shutil.copyfile(runner, directory / 'root-runner.py.txt')
rows = []
for index, argv in enumerate(commands, 1):
    started, timer, timed_out = now(), time.monotonic(), False
    try:
        result = subprocess.run(argv, cwd=ROOT, stdin=subprocess.DEVNULL,
            stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=60, shell=False)
        code, out, err = result.returncode, result.stdout, result.stderr
    except subprocess.TimeoutExpired as error:
        timed_out, code = True, 124
        out, err = error.stdout or b'', error.stderr or b''
    stdout = directory / f'command-{index:02d}.stdout.txt'
    stderr = directory / f'command-{index:02d}.stderr.txt'
    stdout.write_bytes(out)
    stderr.write_bytes(err)
    row = {'index': index, 'argv': argv, 'startedAt': started, 'finishedAt': now(),
        'durationSeconds': round(time.monotonic() - timer, 6), 'exitCode': code, 'timedOut': timed_out,
        'stdout': {'path': str(stdout.relative_to(ROOT)), 'bytes': len(out), 'sha256': sha(out)},
        'stderr': {'path': str(stderr.relative_to(ROOT)), 'bytes': len(err), 'sha256': sha(err)}}
    rows.append(row)
    save(directory / 'command-results.json', {'commands': rows, 'executionEvidenceOnly': True,
        'stoppedAfterFirstNonzeroCommand': code != 0,
        'remainingCommandsNotRun': commands[index:] if code != 0 else [], 'all87': 'HOLD'})
    print(json.dumps({'index': index, 'exitCode': code, 'seconds': row['durationSeconds'],
        'stdoutBytes': len(out), 'stderrBytes': len(err)}), flush=True)
    if code != 0:
        break
after = verify(pins, graph)
save(directory / 'root-after-verification.json', {'checkedAt': now(), 'inputCount': len(pins),
    'wholeGraphFiles': len(graph['files']) if graph else 0,
    'wholeGraphLinks': len(graph['links']) if graph else 0, 'errors': after,
    'allInputsUnchanged': not after, 'noAcceptancePromotion': True})
assert not after, after
print(json.dumps({'inputsUnchanged': len(pins), 'evidenceDirectory': str(directory.relative_to(ROOT))}), flush=True)
