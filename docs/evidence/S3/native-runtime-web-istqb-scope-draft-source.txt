const { Context } = await import(pathToFileURL(`${DSH_WEB_RUNTIME_ROOT}/node_modules/@deepseek-ai/cordis/lib/index.js`));
const key = 'client-connection/browser-session';
const valid = () => ({ kind: 'grant', payload: { version: 1, secret: randomBytes(32).toString('base64url') } });
const digest = value => createHash('sha256').update(value).digest('hex');
const gate = () => { let resolve; const promise = new Promise(r => { resolve = r; }); return { promise, resolve }; };
async function fixture(run) {
  const directory = mkdtempSync(join(tmpdir(), 'df-runtime-credential-independent-'));
  const ctx = new Context(); ctx.logger.exporters?.clear?.();
  const path = join(directory, 'synthetic-private-auth.json'), provider = new FixtureCredentialProvider(ctx, path);
  try { await run({ directory, path, ctx, provider }); }
  finally { await ctx.fiber.dispose(); rmSync(directory, { recursive: true, force: true }); }
}
async function denied(operation) {
  let error; try { await operation(); } catch (caught) { error = caught; }
  assert.ok(error, 'expected denied credential operation');
  assert.equal(error.message === 'FIXTURE_CREDENTIAL_UNAVAILABLE', true, 'expected stable redacted credential error');
}

test('independent synthetic credential file byte cap 1023/1024/1025', async t => {
  for (const bytes of [1023, 1024, 1025]) await t.test(`serialized-bytes-${bytes}`, async () => fixture(async ({ path, provider }) => {
    const json = JSON.stringify(valid()); const body = json + ' '.repeat(bytes - Buffer.byteLength(json));
    writeFileSync(path, body, { mode: 0o600 }); const before = digest(readFileSync(path));
    if (bytes <= 1024) assert.equal((await provider.readRecord(key)).kind, 'grant');
    else await denied(() => provider.readRecord(key));
    assert.equal(digest(readFileSync(path)), before);
  }));
});
test('independent closed persisted credential format partitions', async t => {
  for (const partition of ['version0', 'version2', 'extra-record', 'extra-payload', 'secret42', 'secret44', 'noncanonical']) await t.test(partition, async () => fixture(async ({ path, provider }) => {
    const value = valid();
    if (partition === 'version0') value.payload.version = 0;
    if (partition === 'version2') value.payload.version = 2;
    if (partition === 'extra-record') value.extra = true;
    if (partition === 'extra-payload') value.payload.extra = true;
    if (partition === 'secret42') value.payload.secret = 'A'.repeat(42);
    if (partition === 'secret44') value.payload.secret = 'A'.repeat(44);
    if (partition === 'noncanonical') value.payload.secret = 'A'.repeat(42) + 'B';
    writeFileSync(path, JSON.stringify(value), { mode: 0o600 }); const before = digest(readFileSync(path));
    await denied(() => provider.readRecord(key)); assert.equal(digest(readFileSync(path)), before);
  }));
});

test('independent dispose during pending native credential write denies publication and late callbacks', async () => fixture(async ({ path, ctx, provider }) => {
  const original = valid(); await provider.modifyRecord(key, async () => original); const before = digest(readFileSync(path));
  const entered = gate(), release = gate(); let queuedCalls = 0;
  const pending = provider.modifyRecord(key, async () => { entered.resolve(); await release.promise; return valid(); });
  const pendingDenied = denied(() => pending); await entered.promise;
  const queued = provider.modifyRecord(key, async () => { queuedCalls++; return valid(); });
  const queuedDenied = denied(() => queued); await ctx.fiber.dispose(); release.resolve();
  await Promise.all([pendingDenied, queuedDenied]); assert.equal(queuedCalls, 0); assert.equal(digest(readFileSync(path)), before);
  await denied(() => provider.readRecord(key)); assert.equal(existsSync(`${path}.lock`), false); assert.equal(existsSync(`${path}.tmp`), false);
}));
test('independent two provider contenders do not overwrite an in-flight native signing record', async () => fixture(async ({ path, ctx, provider }) => {
  const original = valid(); await provider.modifyRecord(key, async () => original);
  const secondCtx = new Context(); secondCtx.logger.exporters?.clear?.();
  try {
    const second = new FixtureCredentialProvider(secondCtx, path), entered = gate(), release = gate(); let secondCalls = 0;
    const pending = provider.modifyRecord(key, async current => { entered.resolve(); await release.promise; return current; });
    await entered.promise; await denied(() => second.modifyRecord(key, async () => { secondCalls++; return valid(); }));
    release.resolve(); await pending; assert.equal(secondCalls, 0); assert.equal(digest(JSON.stringify(await provider.readRecord(key))), digest(JSON.stringify(original)));
  } finally { await secondCtx.fiber.dispose(); }
}));
test('independent cleanup I/O failure must retain stable private error projection', async () => fixture(async ({ path, provider }) => {
  await provider.modifyRecord(key, async () => valid()); const before = digest(readFileSync(path));
  await denied(() => provider.modifyRecord(key, async () => { unlinkSync(`${path}.lock`); throw new Error('SYNTHETIC_PRIVATE_CALLBACK_CANARY'); }));
  assert.equal(digest(readFileSync(path)), before);
}));

test('independent credential cleanup cannot remove a replacement lock belonging to another writer', async () => fixture(async ({ path, provider }) => {
  await provider.modifyRecord(key, async () => valid()); const before = digest(readFileSync(path));
  await denied(() => provider.modifyRecord(key, async () => {

const pathsFor = base => ({ base, home: join(base, 'home'), agents: join(base, 'agents'), workspace: join(base, 'workspace'), private: join(base, 'private'), evidence: join(base, 'evidence') });
function pathFixture(run) {
  const directory = mkdtempSync(join(realpathSync(tmpdir()), 'df-runtime-path-independent-'));
  try { run(directory); } finally { rmSync(directory, { recursive: true, force: true }); }
}
test('independent every nested profile symlink refuses before any directory creation or mode change', async t => {
  for (const boundary of ['home', 'home/profiles', 'home/profiles/web']) for (const dangling of [false, true]) await t.test(`${boundary}-${dangling ? 'dangling' : 'existing'}`, () => pathFixture(directory => {
    const base = join(directory, 'fixture'), target = join(directory, 'synthetic-outside-target');
    mkdirSync(base); if (!dangling) mkdirSync(target);
    const entry = join(base, boundary); mkdirSync(join(entry, '..'), { recursive: true }); symlinkSync(target, entry);
    chmodSync(base, 0o750); const beforeMode = lstatSync(base).mode & 0o777;
    assert.throws(() => prepareRuntimeDirectories(pathsFor(base)), { message: 'FIXTURE_PATH_INVALID' });
    assert.equal(readlinkSync(entry), target); assert.equal(lstatSync(base).mode & 0o777, beforeMode);
    for (const name of ['agents', 'workspace', 'private', 'evidence']) assert.equal(existsSync(join(base, name)), false);
    assert.equal(existsSync(join(target, 'package.json')), false); assert.equal(existsSync(join(target, 'cordis.patch.yml')), false);
    assert.equal(existsSync(target), !dangling);
  }));
});
test('independent preparation refuses a symlink ancestor above base and regular file at a directory boundary', async t => {
  for (const partition of ['ancestor', 'regular-file']) await t.test(partition, () => pathFixture(directory => {
    const base = join(directory, 'fixture'), target = join(directory, 'synthetic-outside-target');
    let paths;
    if (partition === 'ancestor') {
      mkdirSync(target); symlinkSync(target, base); paths = pathsFor(join(base, 'nested'));
    } else {
      mkdirSync(base); writeFileSync(join(base, 'home'), 'synthetic-preserved-nondirectory', { mode: 0o600 }); paths = pathsFor(base);
    }
    assert.throws(() => prepareRuntimeDirectories(paths), { message: 'FIXTURE_PATH_INVALID' });
    if (partition === 'ancestor') { assert.equal(readlinkSync(base), target); assert.equal(existsSync(join(target, 'nested')), false); }
    else { assert.equal(readFileSync(join(base, 'home'), 'utf8'), 'synthetic-preserved-nondirectory'); assert.equal(existsSync(join(base, 'agents')), false); }
  }));
});

test('independent genuine scoped WebServer route registration and remount retain original native authentication', async () => {
  const directory = mkdtempSync(join(tmpdir(), 'df-native-scoped-web-independent-')), ctx = new Context(); ctx.logger.exporters?.clear?.();
  const packages = `${DSH_WEB_RUNTIME_ROOT}/node_modules/@deepseek-ai`;
  const { default: SessionStore } = await import(pathToFileURL(`${packages}/dsh-session/lib/index.js`));
  const { default: ApprovalService } = await import(pathToFileURL(`${packages}/dsh-user-approval/lib/index.js`));
  const Connection = await import(pathToFileURL(`${packages}/dsh-client-connection/lib/index.js`));
  const { WebServer } = await import(pathToFileURL(`${packages}/dsh-host-webserver/lib/index.js`));
  try {
    const authPath = join(directory, 'synthetic-auth.json');
    await ctx.plugin({ apply(ownerCtx) { new FixtureCredentialProvider(ownerCtx, authPath); } });
    await ctx.plugin(SessionStore); await ctx.plugin(ApprovalService, { policy: 'never' }); await ctx.plugin(Connection);
    const authBefore = digest(readFileSync(authPath)); let runtime;
    const parent = ctx.plugin({ inject: ['connection', 'sessions', 'approval'], async apply(scopedCtx) {
      runtime = scopedCtx.plugin(createRuntimePlugin({ identity: () => ({ ownerId: 'scoped-synthetic-owner', workspaceId: 'scoped-synthetic-workspace' }), encryptionKey: () => Buffer.alloc(32, 31), hostPermission: () => false }), { databasePath: join(directory, 'ledger.sqlite') });
      await runtime;
    } }); await parent;
    const borrowed = ctx.df; assert.equal(borrowed.diagnostics().supported, true); assert.equal(ctx.sessions.list().length, 0);
    async function mount() {
      let server;
      const fiber = ctx.plugin({ apply(webCtx) { server = new WebServer(webCtx, { host: '127.0.0.1', port: 3080, compression: 'none' }); } });
      await fiber;
      for (let i = 0; i < 30 && !server.prefixes.has('/df-rpc'); i++) await new Promise(resolve => setImmediate(resolve));
      // Only the actual public constructor/register are used. Never load the
      // WebServer class as a plugin or invoke Service.init/listen.
      assert.equal(server.server, undefined); assert.equal(server.prefixes.has('/df-rpc'), true); assert.equal(server.prefixes.has('/api'), true);
      return { fiber, server, route: server.prefixes.get('/df-rpc') };
    }
    const first = await mount();
    let rejected; const missingDependency = ctx.plugin({ inject: ['connection'], apply(scopedCtx) {
      scopedCtx.connection.rpc.handle('/synthetic-missing-dependency', async () => ({}));
    } });
    try { await missingDependency; } catch (error) { rejected = error; }
    assert.equal(rejected?.message.includes('webServer'), true); assert.equal(rejected?.message.includes('without inject'), true);
    await missingDependency.dispose(); assert.equal(first.server.prefixes.has('/synthetic-missing-dependency'), false);
    assert.equal(first.server.prefixes.get('/df-rpc'), first.route);
    await first.fiber.dispose(); assert.equal(first.server.prefixes.has('/df-rpc'), false); assert.equal(first.server.server, undefined);
    assert.equal(borrowed.diagnostics().supported, true);
    const second = await mount(); assert.notEqual(second.route, first.route); assert.equal(first.server.prefixes.has('/df-rpc'), false);
    assert.equal(digest(readFileSync(authPath)), authBefore); assert.equal(ctx.sessions.list().length, 0);
    const shutdown = parent.dispose(); assert.throws(() => borrowed.diagnostics(), { code: 'SERVICE_DISPOSED' });
    await shutdown; assert.equal(second.server.prefixes.has('/df-rpc'), false); assert.equal(second.server.server, undefined); await second.fiber.dispose();
  } finally { await ctx.fiber.dispose(); rmSync(directory, { recursive: true, force: true }); }
});
