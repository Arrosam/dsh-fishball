function barrier() {
  let enter, release;
  const entered = new Promise(resolve => { enter = resolve; });
  const released = new Promise(resolve => { release = resolve; });
  return { entered, release, async wait() { enter(); await released; } };
}

function sqlFacts(path) {
  const db = new DatabaseSync(path, { readOnly: true });
  try {
    const count = table => db.prepare(`SELECT count(*) AS n FROM ${table}`).get().n;
    return { reservations: count('reservations'), commands: count('runtime_commands'), artifacts: db.prepare("SELECT count(*) AS n FROM runtime_records WHERE kind='artifact'").get().n };
  } finally { db.close(); }
}

function byteHashes(directory) {
  const result = {};
  for (const entry of readdirSync(directory, { withFileTypes: true })) {
    const path = join(directory, entry.name);
    if (entry.isDirectory()) for (const [child, value] of Object.entries(byteHashes(path))) result[`${entry.name}/${child}`] = value;
    else if (entry.isFile()) result[entry.name] = hash(readFileSync(path));
  }
  return result;
}

// A separate deterministic adapter is used only for the preserved, advanced
// absent-hooks control. It is registered through the original public LLM API.
class OrdinaryFixtureAdapter extends LlmAdapter {
  constructor(trace) { super(); this.trace = trace; }
  async resolveModel(provider, model) { return { provider, id: model, name: 'Independent ordinary fixture', context: { contextWindow: 65536 } }; }
  async *stream(request) {
    this.trace.push({ maxTokens: request.maxTokens, messages: structuredClone(request.messages) });
    yield { type: 'text-delta', index: 0, text: 'Ordinary advanced fixture output' };
    yield { type: 'usage', usage: { inputTokens: 7, outputTokens: 5, totalTokens: 12 } };
    yield { type: 'finish', reason: { kind: 'stop' } };
  }
}

  return snapshot;
}
