// Root-only passive startup diagnosis. No model, tool, identity or permission API.
import { writeFileSync, renameSync } from 'node:fs';
import { fileURLToPath } from 'node:url';
import { types } from 'node:util';
const base = fileURLToPath(new URL('../', import.meta.url));
const file = `${base}evidence/original-startup-observer.json`;
const state = { stage: 'instrumentation-imported', pid: process.pid, agentRequests: 0, warnings: 0, errors: 0, codes: [], categories: {}, entries: [] };
function write() { writeFileSync(`${file}.tmp`, JSON.stringify(state, null, 2) + '\n', { mode: 0o600 }); renameSync(`${file}.tmp`, file); }
write();
export const name = 'df-ni2-startup-observer';
export const inject = ['loader'];
export function apply(ctx) {
  let active = true;
  const codes = new Set(['CONFIG_INVALID', 'HOST_CAPABILITY_UNSUPPORTED', 'WORKER_SOURCE_UNAVAILABLE', 'WORKER_STORAGE_UNAVAILABLE', 'SETUP_UNAVAILABLE', 'SERVICE_DISPOSED', 'ERR_MODULE_NOT_FOUND', 'EPERM', 'EACCES', 'EADDRINUSE', 'ENOENT']);
  function observe() {
    if (!active) return;
    state.stage = 'original-loader-instrumented';
    state.entries = [...ctx.loader.entries()].map(e => ({ id: typeof e.options.id === 'string' ? e.options.id : null,
      name: /^(@deepseek-ai\/[a-z0-9/-]+|dsh-fishball(\/[a-z0-9-]+)?|cordis:[a-z0-9-]+)$/.test(e.options.name) ? e.options.name : 'local-instrumentation',
      disabled: e.disabled === true, fiberState: e.fiber?.state ?? null, imported: e.moduleNamespace !== undefined }));
    write();
  }
  ctx.logger.exporter({ levels: { default: 2 }, export: event => {
    if (!active || !['warn', 'error'].includes(event.type)) return;
    state[event.type === 'warn' ? 'warnings' : 'errors']++;
    for (const value of event.args ?? []) {
      if (!value || typeof value !== 'object' || types.isProxy(value)) continue;
      const d = Object.getOwnPropertyDescriptor(value, 'code');
      if (d && Object.hasOwn(d, 'value') && codes.has(d.value) && !state.codes.includes(d.value)) state.codes.push(d.value);
      const m = Object.getOwnPropertyDescriptor(value, 'message');
      if (m && Object.hasOwn(m, 'value') && typeof m.value === 'string') {
        const category = m.value.startsWith('permission: composed sandbox and approval defaults match no preset;') ? 'permission-default-unmatched' :
          /^permission: unknown (default )?preset /.test(m.value) ? 'permission-default-unknown' :
          m.value.startsWith('permission: the mounted bash executor does not confine') ? 'permission-executor-unconfined' : null;
        if (category) state.categories[category] = (state.categories[category] ?? 0) + 1;
      }
    }
    observe();
  } });
  ctx.on('internal/status', observe);
  ctx.on('agent/request', () => { state.agentRequests++; observe(); });
  const timer = setInterval(observe, 200); timer.unref();
  ctx.effect(() => () => { active = false; clearInterval(timer); state.stage = 'instrumentation-disposed'; write(); }, 'ni2.startup-observation');
  observe();
}
export default { name, inject, apply };
