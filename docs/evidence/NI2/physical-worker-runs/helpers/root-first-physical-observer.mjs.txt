// Passive physical-package instrumentation. Root alone activates this module.
import { existsSync, readFileSync, writeFileSync, renameSync, unlinkSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { createRequire } from 'node:module';
import { fileURLToPath, pathToFileURL } from 'node:url';
const base=fileURLToPath(new URL('../',import.meta.url));
const requireHost=createRequire(`${base}host/package.json`);
const hostModule=async name=>import(pathToFileURL(requireHost.resolve(`@deepseek-ai/${name}`)).href);
const [{symbols},{Entry},credential,localCredential,jsonl,persistence,loop,connection]=await Promise.all(['cordis','cordis-plugin-loader','dsh-credentials','dsh-credentials-local','dsh-session-persistence-jsonl','dsh-session-persistence','dsh-agent-loop','dsh-client-connection'].map(hostModule));
const requireProfile=createRequire(`${base}home/profiles/web/package.json`);
const [installedModule,testingModule]=await Promise.all(['dsh-fishball/installed','dsh-fishball/testing-provider'].map(name=>import(pathToFileURL(requireProfile.resolve(name)).href)));
const hash=value=>createHash('sha256').update(value).digest('hex');
const original=value=>value?.[symbols.original]??value;
const statePath=`${base}evidence/observer-state.json`, accessPath=`${base}private/native-access.json`, controlPath=`${base}private/lifecycle-command.json`;
export const name='df-ni2-physical-observer';
export const inject=['sessions','agents','llm','tools','connection','webServer','loader','credentials','sessionPersistence','agentLoop'];
export function apply(ctx, config) {
  let active=true,busy=false,setup,runtime;
  const completed=new Set();
  const state={phase:'observer-active',pid:process.pid,setupAvailable:false,runtimeAvailable:false,setupLoads:0,runtimeLoads:0,agentRequests:0,streamEntries:0,streamFinishes:0,textChunks:0,toolAttempts:0,eventCount:0,eventTypes:{},assistantFrames:0,lastUsage:null,request:null,resultDigest:null,persisted:null,withdrawals:[]};
  const entries=()=>[...ctx.loader.entries()];
  const selected=id=>entries().filter(e=>e.options.id===id);
  function facts(){
    const entry=selected('df-installed'), provider=selected('df-testing-provider'), loopEntry=entries().filter(e=>e instanceof Entry&&e.moduleNamespace?.default===loop.default&&!e.disabled&&e.fiber?.state===2);
    return {installed:entry.map(e=>({name:e.options.name,disabled:e.disabled,fiberState:e.fiber?.state??null,moduleExact:e.moduleNamespace?.default===installedModule.default,fiberEntryExact:e.fiber?.entry===e})),testingProvider:provider.map(e=>({name:e.options.name,disabled:e.disabled,fiberState:e.fiber?.state??null,moduleExact:e.moduleNamespace?.default===testingModule.default,fiberEntryExact:e.fiber?.entry===e})),connectionOriginal:ctx.connection instanceof connection.HostConnectionService,operatorOriginal:ctx.connection.operator instanceof connection.OperatorPeer,credentialProviderOriginal:ctx.credentials instanceof credential.CredentialProvider&&ctx.credentials instanceof localCredential.default,credentialPathExact:ctx.credentials.config.path===config.credentialPath,credentialHomeExact:ctx.credentials.config.dshHome===`${base}home`,persistenceOriginal:ctx.sessionPersistence instanceof persistence.SessionPersistence&&ctx.sessionPersistence instanceof jsonl.default,persistenceRootExact:ctx.sessionPersistence.config.root===config.persistenceDirectory,persistenceCompressionExact:ctx.sessionPersistence.config.compression==='none',loopOriginal:ctx.agentLoop instanceof loop.default,loopCapturedBackendExact:loopEntry.length===1&&original(loopEntry[0].ctx.reflect.get('sessionPersistence'))===original(ctx.sessionPersistence),profileExact:ctx.reflect.get('profileContext')?.dir===config.profileDirectory&&ctx.reflect.get('profileContext')?.name==='web'};
  }
  function write(){
    if(!active&&state.phase!=='disposed')return;
    state.sessions=ctx.sessions.list().length;state.agents=ctx.agents.list().map(a=>({status:a.status,idRef:hash(a.id)}));state.providers=ctx.llm.listProviders().length;
    state.databaseExists=existsSync(config.databasePath);state.setupNamespaceExists=existsSync(`${config.databasePath}.df-setup`);state.routes={setup:ctx.webServer.prefixes.has('/df-setup'),runtime:ctx.webServer.prefixes.has('/df-rpc')};state.facts=facts();
    writeFileSync(`${statePath}.tmp`,JSON.stringify(state,null,2)+'\n',{mode:0o600});renameSync(`${statePath}.tmp`,statePath);
  }
  const stale=service=>{try{service?.diagnostics();return 'ACTIVE';}catch(error){return error?.code==='SERVICE_DISPOSED'?'SERVICE_DISPOSED':'UNAVAILABLE';}};
  // Never blocks an original request or changes request/options/provider selection.
  ctx.on('agent/request',()=>{state.agentRequests++;write();});
  ctx.on('agent/assistant-stream',()=>{state.assistantFrames++;});
  ctx.on('session/event',(session,event)=>{if(!session.id.startsWith('df-'))return;state.eventCount++;const key=['user/message','assistant/message','agent/attempt','turn/start','turn/end','step/start','step/end'].includes(event.type)?event.type:'other';state.eventTypes[key]=(state.eventTypes[key]??0)+1;});
  ctx.on('llm/stream',(options,next)=>(async function*(){
    state.streamEntries++;state.request={messageCount:options.messages?.length??null,requestBytes:Array.isArray(options.messages)?Buffer.byteLength(JSON.stringify(options.messages)):null,toolCount:options.tools?.length??0,providerIsSynthetic:options.provider==='df-testing-local',modelIsSynthetic:options.model==='df-testing-worker-v1',maxTokens:options.maxTokens??null,reasoningIsDeterministic:options.reasoningEffort==='deterministic'};write();const textHash=createHash('sha256');let outputBytes=0;
    for await(const chunk of next()){
      if(chunk.type==='text-delta'){state.textChunks++;textHash.update(chunk.text);outputBytes+=Buffer.byteLength(chunk.text);if(outputBytes>128)state.observerError='NI2_OBSERVER_OUTPUT_BOUND';}
      if(chunk.type==='usage'){const u=chunk.usage;state.lastUsage={inputUnits:u.inputTokens,outputUnits:u.outputTokens,totalUnits:u.totalTokens};}
      if(chunk.type==='finish'){state.streamFinishes++;state.resultDigest=textHash.digest('hex');}
      yield chunk;
    }
    write();
  })());
  // Observation only: undefined leaves the product's own monotonic veto intact.
  ctx.tools.guard(()=>{state.toolAttempts++;write();});
  writeFileSync(accessPath,JSON.stringify({url:ctx.connection.authenticatedUrl(`http://127.0.0.1:${ctx.webServer.port}`)})+'\n',{mode:0o600});
  ctx.inject(['dfSetup'],child=>{const value=child.dfSetup;setup=value;state.setupAvailable=true;state.setupLoads++;write();child.effect(()=>()=>{state.setupAvailable=false;state.withdrawals.push({service:'setup',code:stale(value)});write();},'ni2.observe-setup');});
  ctx.inject(['df'],child=>{const value=child.df;runtime=value;state.runtimeAvailable=true;state.runtimeLoads++;write();child.effect(()=>()=>{state.runtimeAvailable=false;state.withdrawals.push({service:'runtime',code:stale(value)});write();},'ni2.observe-runtime');});
  async function readPersisted(){
    const stored=await ctx.sessionPersistence.list();if(stored.length!==1)throw new Error('NI2_PERSISTED_COUNT');
    const handle=await ctx.sessionPersistence.open(stored[0].header.id,'read');
    try{const {events}=await handle.read();const users=events.filter(e=>e.type==='user/message'),assistant=events.filter(e=>e.type==='assistant/message');const text=message=>(message.data.message?.content??message.data.content??[]).filter(c=>c.type==='text').map(c=>c.text).join('');
      state.persisted={count:1,idRef:hash(handle.id),events:events.length,userMessages:users.length,assistantMessages:assistant.length,inputDigest:users.length===1?hash(text(users[0])):null,resultDigest:assistant.length===1?hash(text(assistant[0])):null};
    }finally{await handle.close();}
  }
  async function command(value){
    state.lifecycle={id:value.id,operation:value.operation,result:'pending'};write();
    if(value.operation==='persisted-read'){await readPersisted();}
    else if(value.operation==='reload10'){
      const chosen=selected('df-installed');if(chosen.length!==1||chosen[0].options.name!=='dsh-fishball/installed')throw new Error('NI2_ENTRY_SELECTION');
      const before={agentRequests:state.agentRequests,streamEntries:state.streamEntries,toolAttempts:state.toolAttempts};state.lifecycle.cycles=[];
      for(let index=0;index<10&&active;index++){
        const priorSetup=setup,priorRuntime=runtime;const stopping=chosen[0].update({disabled:true});const setupCode=stale(priorSetup),runtimeCode=stale(priorRuntime);await stopping;await ctx.loader.await();await new Promise(resolve=>setImmediate(resolve));
        const withdrawn=!state.setupAvailable&&!state.runtimeAvailable&&!ctx.webServer.prefixes.has('/df-setup')&&!ctx.webServer.prefixes.has('/df-rpc');
        await chosen[0].update({disabled:false});await ctx.loader.await();const deadline=Date.now()+10000;
        while(active&&(!state.setupAvailable||!state.runtimeAvailable)&&Date.now()<deadline)await new Promise(resolve=>setTimeout(resolve,25));
        const cycle={index:index+1,setupCode,runtimeCode,withdrawn,setupPhase:setup.diagnostics().phase,sessions:ctx.sessions.list().length,agentRequests:state.agentRequests-before.agentRequests,streamEntries:state.streamEntries-before.streamEntries,toolAttempts:state.toolAttempts-before.toolAttempts};state.lifecycle.cycles.push(cycle);write();
        if(setupCode!=='SERVICE_DISPOSED'||runtimeCode!=='SERVICE_DISPOSED'||!withdrawn||cycle.setupPhase!=='ready'||cycle.sessions||cycle.agentRequests||cycle.streamEntries||cycle.toolAttempts)throw new Error('NI2_RELOAD_ORACLE');
      }
      if(state.lifecycle.cycles.length!==10)throw new Error('NI2_RELOAD_INCOMPLETE');
    }else throw new Error('NI2_OBSERVER_COMMAND');
    state.lifecycle.result='passed';write();
  }
  const timer=setInterval(()=>{if(!active)return;try{write();}catch{state.observerError='NI2_OBSERVATION_UNAVAILABLE';}if(busy||!existsSync(controlPath))return;let value;try{value=JSON.parse(readFileSync(controlPath,'utf8'));}catch{return;}
    if(Object.keys(value??{}).sort().join(',')!=='id,operation'||typeof value.id!=='string'||!/^[a-z0-9-]{1,40}$/.test(value.id)||!['persisted-read','reload10'].includes(value.operation)||completed.has(value.id))return;completed.add(value.id);busy=true;
    command(value).catch(()=>{state.lifecycle={id:value.id,operation:value.operation,result:'failed',code:'NI2_OBSERVER_ORACLE_FAILED'};try{write();}catch{}}).finally(()=>{try{const current=JSON.parse(readFileSync(controlPath,'utf8'));if(current.id===value.id)unlinkSync(controlPath);}catch{}busy=false;});
  },200);
  ctx.effect(()=>()=>{clearInterval(timer);active=false;state.phase='disposed';try{unlinkSync(accessPath);}catch{}try{write();}catch{}},'ni2.physical-observer');write();
}
