import fs from 'node:fs';
import crypto from 'node:crypto';
import {fileURLToPath} from 'node:url';
const base=fileURLToPath(new URL('../',import.meta.url));
const origin='http://127.0.0.1:3080';
const url=new URL(JSON.parse(fs.readFileSync(base+'private/native-access.json')).url);
if(url.origin!==origin||url.pathname!=='/'||url.searchParams.size!==1||!url.searchParams.has('token'))throw Error('NI2_AUTH_ADDRESS');
const init=await fetch(url,{redirect:'manual',signal:AbortSignal.timeout(10000)});
if(init.status!==303)throw Error('NI2_AUTH_EXCHANGE');
let cookie=init.headers.get('set-cookie')?.split(';')[0];if(!cookie)throw Error('NI2_AUTH_COOKIE');
const taskRef=crypto.createHash('sha256').update('ni2-physical-one-task').digest('hex');
const response=await fetch(origin+'/df-rpc/v1/status',{method:'POST',headers:{origin,'content-type':'application/json',cookie},body:JSON.stringify({type:'client-request',rpcId:'ni2-readonly-diagnosis',method:'v1/status',payload:{taskRef}}),signal:AbortSignal.timeout(10000)});
cookie=undefined;
if(response.status!==200)throw Error('NI2_READ_HTTP');
const value=(await response.json()).result;if(value?.ok!==true)throw Error('NI2_READ_REFUSED');
const v=value.value,pick=(r,keys)=>Object.fromEntries(keys.filter(k=>Object.hasOwn(r,k)).map(k=>[k,r[k]]));
const report={kind:'original-owner-readonly-first-chain',task:pick(v.task,['taskRef','status','version','authEpoch']),keys:Object.keys(v),actions:v.actions?.map(r=>pick(r,['actionRef','actionDigest','state','outcome','claimRef','generationRef','usageState','usageUnits'])),workers:v.workers?.map(r=>pick(r,['actionRef','claimRef','generationRef','phase','reservationRef','cursor','executionRef'])),executions:v.executions?.map(r=>pick(r,['executionRef','state','claimRef','generationRef','cursor','dataDigest','contextRecordRef'])),artifacts:v.artifacts?.length};
fs.writeFileSync(base+'evidence/root-first-chain-readonly.json',JSON.stringify(report,null,2)+'\n',{mode:0o600});
console.log(JSON.stringify(report));
